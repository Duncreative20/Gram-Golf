## Parity — script dump token theo variant (dùng cho `src/audit/figma/*.json`, spec §7e)

Thay `JOBS = [[nodeId, /regex lọc variant/], …]` ở đầu; 1 call ≤19 KB → lọc Size/State, mỗi call 1–3 set. Kết quả dán vào `packages/ds/src/audit/figma/<set>.json`, thêm entry `CONFIG` trong `figmaParity.test.ts`.

```js
const JOBS = [['260:4825', /Size=Medium/]];
// Template use_figma: thay JOBS. Mỗi job = [nodeId, keepRegex]. Output ≤19KB.
const cache = {}; async function vname(id){ if(cache[id]===undefined){ const v = await figma.variables.getVariableByIdAsync(id); cache[id] = v ? v.name : '?'; } return cache[id]; }
const sc = {}; async function sname(id){ if(typeof id!=='string'||!id) return null; if(sc[id]===undefined){ const s = await figma.getStyleByIdAsync(id); sc[id]= s? s.name : '?'; } return sc[id]; }
function hex(p){ if(!p||p.type!=='SOLID') return p? p.type : null; const c=p.color; const h=v=>Math.round(v*255).toString(16).padStart(2,'0'); return '#'+h(c.r)+h(c.g)+h(c.b)+(p.opacity!==undefined&&p.opacity<1?('@'+Math.round(p.opacity*100)):''); }
async function walk(n, out){
  if(n.visible===false) return;
  if(n.type==='INSTANCE'){ const mc = await n.getMainComponentAsync(); if(mc && mc.remote) { out.add('remote:'+mc.name); return; } }
  const bv = n.boundVariables||{};
  const fills = Array.isArray(n.fills)? n.fills.filter(f=>f.visible!==false):[];
  for(const f of fills){ const b=f.boundVariables&&f.boundVariables.color; out.add((n.type==='TEXT'?'text:':'fill:')+(b? await vname(b.id): hex(f))); }
  const strokes = Array.isArray(n.strokes)? n.strokes.filter(f=>f.visible!==false):[];
  for(const f of strokes){ const b=f.boundVariables&&f.boundVariables.color; out.add('stroke:'+(b? await vname(b.id):hex(f))); if(bv.strokeWeight) out.add('strokeW:'+await vname(bv.strokeWeight.id)); }
  if(n.type==='TEXT'){ out.add('style:'+(await sname(n.textStyleId)||'mixed')); }
  if(n.layoutMode && n.layoutMode!=='NONE'){ for(const k of ['paddingTop','paddingRight','paddingBottom','paddingLeft','itemSpacing']){ const b=bv[k]; if(b) out.add(k.replace('padding','pad').replace('itemSpacing','gap')+':'+await vname(b.id)); } }
  for(const k of ['topLeftRadius','width','height','minWidth','minHeight']){ const b=bv[k]; if(b) out.add((k==='topLeftRadius'?'radius':k)+':'+await vname(b.id)); }
  if(n.effectStyleId) out.add('effect:'+(await sname(n.effectStyleId)));
  else if(Array.isArray(n.effects)) for(const e of n.effects){ if(e.visible===false) continue; const b=e.boundVariables&&e.boundVariables.color; out.add('effect#:'+e.type+(b?':'+await vname(b.id):'')); }
  if('children' in n) for(const c of n.children) await walk(c,out);
}
const res = {};
for(const [id, keep] of JOBS){
  const set = await figma.getNodeByIdAsync(id);
  const variants = set.type==='COMPONENT_SET'? set.children : [set];
  const sets = {};
  for(const v of variants){ if(!keep.test(v.name)) continue; const out=new Set(); await walk(v,out); sets[v.name]=out; }
  const names = Object.keys(sets); let common = null;
  for(const nm of names){ common = common===null? new Set(sets[nm]) : new Set([...common].filter(x=>sets[nm].has(x))); }
  const vr = {}; for(const nm of names){ vr[nm] = [...sets[nm]].filter(x=>!common.has(x)); }
```
